{{- /* Menu z data/menu.yaml jako dane. Wejscie: nazwa listy (string) - "pozycje" (menu glowne, dwa poziomy: pozycje i podmenu) albo lista plaska "dodatkowe", "stopka", "stopka_dolna" (bez podmenu). Wynik: lista dict "etykieta" (pusta = tytul strony, potem adres), "href", "strona" (Page albo false), "zewnetrzny" (bool), "dzieci" (lista tak samo zbudowanych dict; w liscie plaskiej pusta). Strony ukryte (draft) sa pomijane; strona, ktorej nie ma, to blad budowania (partial cel-linku.html). Wywolywany przez partialCached z nazwa listy jako kluczem - ten sam wynik dla menu, szuflady, stopki i okruszkow. */ -}} {{- $nazwa := . -}} {{- $glowne := eq $nazwa "pozycje" -}} {{- $opisy := dict "dodatkowe" "pozycje dodatkowe" "stopka" "stopka" "stopka_dolna" "dolna linia stopki" -}} {{- $zrodlo := "Menu (data/menu.yaml)" -}} {{- if not $glowne }}{{ $zrodlo = printf "Menu (data/menu.yaml, %s)" (index $opisy $nazwa | default $nazwa) }}{{ end -}} {{- $wynik := slice -}} {{- range $i, $poz := index (hugo.Data.menu | default dict) $nazwa -}} {{- $gdzie := printf "%s: pozycja „%s”" $zrodlo ($poz.etykieta | default (printf "nr %d" (add $i 1))) -}} {{- $dzieci := slice -}} {{- if $glowne -}} {{- range $j, $pod := $poz.podmenu -}} {{- $gdziePod := printf "%s, podmenu „%s”" $gdzie ($pod.etykieta | default (printf "nr %d" (add $j 1))) -}} {{- with partial "inline/menu-pozycja.html" (dict "pozycja" $pod "gdzie" $gdziePod) }}{{ $dzieci = $dzieci | append . }}{{ end -}} {{- end -}} {{- end -}} {{- with partial "inline/menu-pozycja.html" (dict "pozycja" $poz "gdzie" $gdzie "dzieci" $dzieci "podmenu" $glowne) }}{{ $wynik = $wynik | append . }}{{ end -}} {{- end -}} {{- return $wynik -}} {{- define "partials/inline/menu-pozycja.html" -}} {{- /* Jedna pozycja menu. Wejscie: dict "pozycja" (z data/menu.yaml), "gdzie", "dzieci" (gotowe podmenu), "podmenu" (bool - pozycja moze miec podmenu). Wynik: dict pozycji albo false (strona ukryta); brak celu = blad. */ -}} {{- $poz := .pozycja -}} {{- $cel := partial "cel-linku.html" (dict "strona" $poz.strona "link_zewnetrzny" $poz.link_zewnetrzny "gdzie" .gdzie) -}} {{- $dzieci := .dzieci | default slice -}} {{- $wynik := false -}} {{- if or $cel.href $dzieci -}} {{- $etykieta := $poz.etykieta -}} {{- if and (not $etykieta) $cel.strona }}{{ $etykieta = $cel.strona.LinkTitle }}{{ end -}} {{- $wynik = dict "etykieta" ($etykieta | default $cel.href) "href" $cel.href "strona" $cel.strona "zewnetrzny" (and $cel.href (not $cel.strona)) "dzieci" $dzieci -}} {{- else if not $cel.ukryta -}} {{- errorf "%s nie ma ani strony, ani linku zewnętrznego%s. Uzupełnij albo usuń ją w panelu /admin/ (Menu)." .gdzie (cond .podmenu ", ani podmenu" "") -}} {{- end -}} {{- return $wynik -}} {{- end -}}
