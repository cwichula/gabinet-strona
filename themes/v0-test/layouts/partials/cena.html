Wejscie: pozycja z data/cennik.yaml. Brak cena_do = jedna cena. "- . ,") "," " " }}{{ end -}} {{- $napisy = $napisy | append $n -}} {{- end -}} {{- $tekst = printf "%s zł" (delimit $napisy "–") -}} {{- end -}} {{- return $tekst -}}
