{{- /* Bloki z front matter "sekcje" (docs/KONTRAKT-BLOKOW.md) w sekcjach makiety v1:
...

Tlo (wariant), szerokosc (waska) i kotwica sekcji - z jej pierwszego bloku; kotwica bloku dolaczonego trafia na jego div. Gdy sekcja zaczyna sie blokiem z kolumna obok (tekst ze zdjeciem, tekst z ramka z boku, kontakt z ramka z boku) albo ma dolaczony blok z ramka z boku, kolejne bloki dolaczone trafiaja do jego kolumny tekstu (znacznik w partialu bloku) - jak w makiecie v1, gdzie kroki, karta rezerwacji czy cytat stoja obok zdjecia albo ramki. "tresc") = blad. */ -}} {{- $strona := . -}} {{- $plik := "Strona" }}{{ with .File }}{{ $plik = .Path }}{{ end -}} {{- $sekcje := .Params.sekcje | default slice -}} {{- /* pamiec blokow na czas jednego renderowania strony ("raz na strone": lightbox galerii, wyszukiwarka cennika, kotwice grupa-*). Zmień ją w panelu /admin/." $plik . (add $nr 1) }}{{ end -}} {{- if or (in $zarezerwowane .) (findRE `^(grupa|podmenu)-|^f[0-9]+-` .) -}} {{- errorf "%s: kotwica %q (sekcja nr %d) jest zajęta przez szablon strony. Wybierz inną nazwę w panelu /admin/." $plik . (add $nr 1) -}} {{- end -}} {{- $kotwice = merge $kotwice (dict . true) -}} {{- end -}} {{- if not (and $typ (templates.Exists (printf "partials/blocks/%s.html" $typ))) -}} {{- warnf "%s: sekcja nr %d ma nieznany typ bloku %q i została pominięta. Dozwolone typy opisuje docs/KONTRAKT-BLOKOW.md." $plik (add $nr 1) $typ -}} {{- else -}} {{- $el := dict "blok" $b "nr" $nr -}} {{- $poprzedniHero := false }}{{ with $biezaca }}{{ if eq (index . 0).blok.type "hero" }}{{ $poprzedniHero = true }}{{ end }}{{ end -}} {{- if and $b.polacz $biezaca (ne $typ "hero") (not $poprzedniHero) -}} {{- $biezaca = $biezaca | append $el -}} {{- else -}} {{- with $biezaca }}{{ $grupy = $grupy | append (dict "bloki" .) }}{{ end -}} {{- $biezaca = slice $el -}} {{- end -}} {{- end -}} {{- end -}} {{- with $biezaca }}{{ $grupy = $grupy | append (dict "bloki" .) }}{{ end -}} {{- /* sekcje */ -}} {{- $waskie := slice "faq" "formularz" "rezerwacja" -}} {{- $znacznik := "" -}} {{- $doKolumny := slice "tekst" "ramka" "kroki" "lista" "cytat" "przycisk" "rezerwacja" "godziny" -}} {{- range $g := $grupy -}} {{- $grupa := $g.bloki -}} {{- $pierwszy := (index $grupa 0).blok -}} {{- $html := "" }}{{ $zwarta := eq $pierwszy.type "pasek_zaufania" -}} {{- range $i, $el := $grupa -}} {{- if eq $el.blok.type "spis" }}{{ $zwarta = true }}{{ end -}} {{- $out := partial (printf "blocks/%s.html" $el.blok.type) (dict "blok" $el.blok "strona" $strona "nr" $el.nr "gora" (eq $i 0) "priorytet" (eq $el.nr $nrZdjecia) "pamiec" $pamiec "gdzie" $plik) | string | strings.TrimSpace -}} {{- if $out -}} {{- if not $html }}{{ $html = $out }}{{ else -}} {{- $id := "" }}{{ with $el.blok.kotwica }}{{ $id = printf ` id="%s"` (. | urlize) }}{{ end -}} {{- $dalej := printf "\n
\n%s\n
" $id $out -}} {{- /* blok dolaczony: do otwartej kolumny tekstu albo pod spodem. Kolumne otwiera pierwszy blok sekcji (tekst ze zdjeciem, tekst z ramka z boku, kontakt z ramka) albo dolaczony blok z ramka z boku (tresc_z_bokiem, kontakt) - wtedy zamyka poprzednia. Do kolumny trafiaja tylko bloki "waskie" ($doKolumny); blok szeroki (karty, tabela, powiazane strony, cta, tekst ze zdjeciem...) staje pod spodem i zamyka kolumne. */ -}} {{- if strings.Contains $out $znacznik -}} {{- if in (slice "tresc_z_bokiem" "kontakt") $el.blok.type -}} {{- $html = printf "%s%s" (replace $html $znacznik "") $dalej -}} {{- else -}} {{- $html = printf "%s%s" (replace $html $znacznik "") (replace $dalej $znacznik "") -}} {{- end -}} {{- else if and (strings.Contains $html $znacznik) (in $doKolumny $el.blok.type) -}} {{- $html = replace $html $znacznik (printf "%s%s" $dalej $znacznik) -}} {{- else -}} {{- $html = printf "%s%s" (replace $html $znacznik "") $dalej -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $html = replace $html $znacznik "" -}} {{- if $html -}} {{- if eq $pierwszy.type "hero" }} {{ $html | safeHTML }} {{- else }} {{- $klasy := slice "section" }} {{- if eq $pierwszy.wariant "wyrozniony" }}{{ $klasy = $klasy | append "section--surface" }}{{ end }} {{- if $zwarta }}{{ $klasy = $klasy | append "section--tight" }}{{ end }} {{- $waska := or $pierwszy.waska (in $waskie $pierwszy.type) }}
{{ $html | safeHTML }}
{{- end }} {{- end -}} {{- end -}}
