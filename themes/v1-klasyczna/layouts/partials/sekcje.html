(add $nr 1) }}{{ end -}} {{- $kotwice = merge $kotwice (dict . true) -}} {{- end -}} {{- if not (and $typ (templates.Exists (printf "partials/blocks/%s.html" $typ))) -}} {{- warnf "%s: sekcja nr %d ma nieznany typ bloku %q i została pominięta. \n%s\n
" $html $id $out -}} {{- end -}} {{- end -}} {{- end -}} {{- if $html -}} {{- if eq $pierwszy.type "hero" }} {{ $html | safeHTML }} {{- else }} {{- $klasy := slice "section" }} {{- if eq $pierwszy.wariant "wyrozniony" }}{{ $klasy = $klasy | append "section--surface" }}{{ end }} {{- if eq $pierwszy.type "pasek_zaufania" }}{{ $klasy = $klasy | append "section--tight" }}{{ end }}
{{ $html | safeHTML }}
